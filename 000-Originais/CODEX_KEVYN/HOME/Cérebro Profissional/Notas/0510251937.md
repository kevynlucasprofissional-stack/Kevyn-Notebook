-- 1) Extensão (só cria se não existir)

create extension if not exists vector;

  

-- 2) Tabela (768 dim, p/ Gemini ou modelos 768)

create table if not exists esbelta (

  id bigserial primary key,

  content  text,

  metadata jsonb,

  embedding vector(768)

);

  

-- 3) Índice para busca por similaridade (cosseno)

create index if not exists esbelta_embedding_idx

  on esbelta using ivfflat (embedding vector_cosine_ops) with (lists = 100);

  

-- 4) Função de busca (aponta para a tabela esbelta)

create or replace function match_documents(

  query_embedding vector(768),

  match_count int default 10,

  filter jsonb default '{}'::jsonb

) returns table (

  id bigint,

  content text,

  metadata jsonb,

  similarity float

)

language sql stable

as $$

  select

    id,

    content,

    metadata,

    1 - (embedding <=> query_embedding) as similarity

  from esbelta

  where metadata @> filter

  order by embedding <=> query_embedding

  limit coalesce(match_count, 10);

$$;

analyze esbelta;
